# C11-C 2.19.1 — Root Cleanup Commands

The final repository keeps only project entrypoints, gate files, README and agent/operator guidance in the root.

Run these commands **from the repository root** after applying the documentation/reorganization overlay. They move only the known historical/operational files listed here; `GeneradorMaestro.gd`, `Main.tscn`, `build_factory.py`, `release_gate.py`, `project.godot` and standard root metadata are intentionally untouched.

```powershell
$root = (Get-Location).Path

$moves = @{
  'C11-C_2.18.x_FREEZE_ACCEPTANCE.md' = 'docs/history/c11c/releases/C11-C_2.18.x_FREEZE_ACCEPTANCE.md'
  'C11C_2.18.5_FINAL_FREEZE_CANDIDATE.md' = 'docs/history/c11c/releases/C11C_2.18.5_FINAL_FREEZE_CANDIDATE.md'
  'C11C_2.18.6_TEST_HOTFIX.md' = 'docs/history/c11c/releases/C11C_2.18.6_TEST_HOTFIX.md'
  'C11C_2.18.7_PRODUCER_AVI_CAPTURE_HOTFIX.md' = 'docs/history/c11c/releases/C11C_2.18.7_PRODUCER_AVI_CAPTURE_HOTFIX.md'

  'C11C_SUITE_0.1.2_OVERLAY.md' = 'docs/history/c11c/releases/C11C_SUITE_0.1.2_OVERLAY.md'
  'CHANGELOG_C11-C_2.19.0.md' = 'docs/history/c11c/releases/CHANGELOG_C11-C_2.19.0.md'
  'CHANGELOG_C11-C_2.19.1.md' = 'docs/history/c11c/releases/CHANGELOG_C11-C_2.19.1.md'
  'CHANGELOG_C11-C_2.19.2.md' = 'docs/history/c11c/releases/CHANGELOG_C11-C_2.19.2.md'
  'README.txt' = 'docs/history/root/README_legacy.txt'
  'build_factory.py' = 'build_factory.py'
}

# The 2.15/2.16 historical documents are moved as well when present in the local root.
$extra = @{
  'C11-C_2.15.1_ART_DIRECTION_CONTRACT_REPAIR.md' = 'docs/history/c11c/releases/C11-C_2.15.1_ART_DIRECTION_CONTRACT_REPAIR.md'
  'C11-C_2.16_FINAL_FREEZE.md' = 'docs/history/c11c/releases/C11-C_2.16_FINAL_FREEZE.md'
}
foreach ($k in $extra.Keys) { $moves[$k] = $extra[$k] }

foreach ($srcRel in $moves.Keys) {
  if ($srcRel -eq $moves[$srcRel]) { continue }
  $src = Join-Path $root $srcRel
  if (-not (Test-Path -LiteralPath $src -PathType Leaf)) { continue }
  $dst = Join-Path $root $moves[$srcRel]
  New-Item -ItemType Directory -Force -Path (Split-Path -Parent $dst) | Out-Null
  Move-Item -LiteralPath $src -Destination $dst -Force
}

# Operational root launchers.
$toolMoves = @{
  'clean-godot.ps1' = 'tools/maintenance/clean-godot.ps1'
  'clean-videos.ps1' = 'tools/maintenance/clean-videos.ps1'
  'Make_zip.ps1' = 'tools/maintenance/Make_zip.ps1'
  'run_c11c_full_art_direction_corpus.ps1' = 'tools/qa/c11/run_c11c_full_art_direction_corpus.ps1'
  'run_c11c_full_art_direction_corpus.cmd' = 'tools/qa/c11/run_c11c_full_art_direction_corpus.cmd'
  'resume_c11c_full_art_direction_corpus_after_longform.ps1' = 'tools/qa/c11/resume_c11c_full_art_direction_corpus_after_longform.ps1'
  'resume_c11c_full_art_direction_corpus_after_longform.cmd' = 'tools/qa/c11/resume_c11c_full_art_direction_corpus_after_longform.cmd'
}
foreach ($srcRel in $toolMoves.Keys) {
  $src = Join-Path $root $srcRel
  if (-not (Test-Path -LiteralPath $src -PathType Leaf)) { continue }
  $dst = Join-Path $root $toolMoves[$srcRel]
  New-Item -ItemType Directory -Force -Path (Split-Path -Parent $dst) | Out-Null
  Move-Item -LiteralPath $src -Destination $dst -Force
}

# D transfer documents from root to their canonical current locations.
$dMoves = @{
  'C11-D_ROADMAP_V1.0_STATELESS.md' = 'docs/current/d/C11-D_ROADMAP_V1.0_STATELESS.md'
  'MASTER_HANDOVER_C11D_V1.0_STATELESS.md' = 'docs/master-prompts/MASTER_HANDOVER_C11D_V1.0_STATELESS.md'
  'START_PROMPT_C11D_V1.0_STATELESS.md' = 'docs/master-prompts/START_PROMPT_C11D_V1.0_STATELESS.md'
  'NEXT-PROMT.TXT' = 'docs/master-prompts/NEXT_PROMPT_C11D_V1.0_STATELESS.txt'
}
foreach ($srcRel in $dMoves.Keys) {
  $src = Join-Path $root $srcRel
  if (-not (Test-Path -LiteralPath $src -PathType Leaf)) { continue }
  $dst = Join-Path $root $dMoves[$srcRel]
  New-Item -ItemType Directory -Force -Path (Split-Path -Parent $dst) | Out-Null
  Move-Item -LiteralPath $src -Destination $dst -Force
}

# Overlay metadata belongs to legacy evidence, not the source root.
foreach ($name in @('OVERLAY_MANIFEST.json','OVERLAY_SHA256.json')) {
  $src = Join-Path $root $name
  if (Test-Path -LiteralPath $src -PathType Leaf) {
    $dst = Join-Path $root "artifacts/legacy/c11c_overlay_metadata/$name"
    New-Item -ItemType Directory -Force -Path (Split-Path -Parent $dst) | Out-Null
    Move-Item -LiteralPath $src -Destination $dst -Force
  }
}

Write-Host '[C11-C] Root cleanup completed. Run verify_repository_layout.ps1 and the full test runner.' -ForegroundColor Green
```

Do **not** remove `.githooks` if it is version-controlled. Do not manually delete `.godot`, `.pytest_cache`, `*.import` or `*.uid` as part of this source reorganization; those are local/editor state and can be regenerated according to the project's normal maintenance procedure.
