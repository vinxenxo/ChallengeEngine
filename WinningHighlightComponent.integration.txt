       func _build_winning_highlight_rects() -> Array[Rect2]:
           var rects: Array[Rect2] = []
           if object_sprite == null:
               return rects

           var object_rect := _sprite_visual_rect_in_ui(object_sprite)
           if object_rect.size.x > 0.0 and object_rect.size.y > 0.0:
               rects.append(object_rect)

           if target_sprite != null and target_sprite.visible:
               var target_rect := _sprite_visual_rect_in_ui(target_sprite)
               if target_rect.size.x > 0.0 and target_rect.size.y > 0.0:
                   rects.append(target_rect)

           return rects

       func _sprite_visual_rect_in_ui(sprite: Sprite2D) -> Rect2:
           if sprite == null or sprite.texture == null or presentation_ui_root == null:
               return Rect2()

           var texture_size: Vector2 = sprite.texture.get_size() * sprite.global_scale
           var top_left_global: Vector2 = sprite.global_position - (texture_size * 0.5)
           var bottom_right_global: Vector2 = sprite.global_position + (texture_size * 0.5)
           var top_left_ui: Vector2 = presentation_ui_root.to_local(top_left_global)
           var bottom_right_ui: Vector2 = presentation_ui_root.to_local(bottom_right_global)
           return Rect2(top_left_ui, bottom_right_ui - top_left_ui)

   IMPORTANTE:
   - La función anterior es puramente geométrica/presentacional.
   - No accede a RNG.
   - No calcula distancia, colisión, score ni winning_frame.
   - No modifica SimulationResult.
